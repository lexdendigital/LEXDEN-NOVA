// LEXDEN NOVA — Firestore security rules.
//
// This file only defines the pieces referenced directly by index.html's
// comments (the "orders" collection contract). If your project already has
// rules for other collections (reviews, leads, clicks, products, etc.),
// merge this "orders" block into your existing rules.rules file rather
// than replacing the whole thing — deploy with:
//   firebase deploy --only firestore:rules

service cloud.firestore {
  match /databases/{database}/documents {

    // Must exactly match ADMIN_EMAIL in index.html.
    function isAdmin() {
      return request.auth != null && request.auth.token.email == "ruthanselem2@gmail.com";
    }

    // Orders are written ONLY by the /api/verify-paystack serverless
    // function, via the Firebase Admin SDK — Admin SDK calls bypass these
    // rules entirely, so there is intentionally no "allow create/update"
    // clause below for regular clients. A shopper can read their own
    // orders (uid match) or the admin can read all of them; nobody can
    // write from the browser, which is what stops a shopper from
    // fabricating a "paid" order by calling Firestore directly.
    match /orders/{orderId} {
      allow read: if request.auth != null &&
                     (resource.data.uid == request.auth.uid || isAdmin());
      allow write: if false;
    }
  }
}
